W_chip	10[mm]	Chip width
t_chip	0.5[mm]	Chip thickness
L_TIM	10[mm]	TIM length
W_TIM	10[mm]	TIM width
t_TIM	0.1[mm]	TIM thickness
k_TIM	3[W/(m*K)]	Effective TIM thermal conductivity
rho_TIM	2500[kg/m^3]	Assumed TIM density
Cp_TIM	1000[J/(kg*K)]	Assumed TIM heat capacity
L_spreader	12[mm]	Heat spreader length
W_spreader	12[mm]	Heat spreader width
t_spreader	1[mm]	Heat spreader thickness
L_plate	30[mm]	Cold plate length
W_plate	30[mm]	Cold plate width
t_plate	3[mm]	Cold plate thickness
N_ch	10	Number of channels
L_ch	25[mm]	Channel length
w_ch	0.5[mm]	Channel width
h_ch	0.5[mm]	Channel height
pitch_ch	1[mm]	Channel center-to-center spacing
t_bottom	1.25[mm]	Copper thickness below channel
z_ch	t_bottom	Channel bottom position
Q_chip	100[W]	Total chip heat generation
Vdot_total	0.2[l/min]	Total coolant flow through all channels
T_in	25[degC]	Coolant inlet temperature
p_out	0[Pa]	Outlet gauge pressure
T_limit	85[degC]	Project chip temperature limit
A_chip	L_chip*W_chip	Chip area
Vol_chip	L_chip*W_chip*t_chip	Chip volume
q_chip	Q_chip/A_chip	Chip surface heat flux
Qvol_chip	Q_chip/Vol_chip	Chip volumetric heat generation
Aflow_total	N_ch*w_ch*h_ch	Total channel flow area
Vdot_ch	Vdot_total/N_ch	Flow rate per channel
u_in	Vdot_total/Aflow_total	Average channel inlet velocity
Dh	2*w_ch*h_ch/(w_ch+h_ch)	Hydraulic diameter
Rth_limit	(T_limit-T_in)/Q_chip	Target system thermal resistance
x_ch	(L_plate-L_ch)/2	Channel x position
y_ch1	(W_plate-((N_ch-1)*pitch_ch+w_ch))/2	First channel y position
x_spreader	(L_plate-L_spreader)/2	Spreader x position
y_spreader	(W_plate-W_spreader)/2	Spreader y position
z_spreader	t_plate	Spreader z position
x_TIM	(L_plate-L_TIM)/2	TIM x position
y_TIM	(W_plate-W_TIM)/2	TIM y position
z_TIM	t_plate+t_spreader	TIM z position
x_chip	(L_plate-L_chip)/2	Chip x position
y_chip	(W_plate-W_chip)/2	Chip y position
z_chip	t_plate+t_spreader+t_TIM	Chip z position
